/**
 * [LicenseGuard Server v4.6 - Security & Reset Optimized]
 * - 복합 식별자(Composite ID) 및 보안 토큰 검증 지원
 * - 기기 변경 시 전용 RESET 칸(Col 8)을 활용한 쿨다운 정밀 제어
 * - 하드웨어 업그레이드 시 연락처 기반 자동 승인 (7일 쿨다운 적용)
 */

// --- 설정 (시트 구조에 맞게 수정 가능) ---
const SECURITY_TOKEN = "EzImpo_Secure_Handshake_Token_v3_X9Z";
const COL = {
  KEY: 1,
  PIN: 2,
  COMPANY: 3,
  USER_INFO: 4,
  MACHINE_ID: 5,
  STATUS: 7,
  RESET: 8,    // 기기 변경일 기록 전용
  CHECKIN: 9,  // 마지막 접속일 기록 전용
  VERSION: 12
};

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const ws = ss.getSheetByName("Licenses");
  const logWs = ss.getSheetByName("InstallLogs");
  
  try {
    const data = JSON.parse(e.postData.contents);
    
    // 0. 보안 토큰 검증
    if (data.token !== SECURITY_TOKEN) {
      return response({ valid: false, message: "인가되지 않은 요청입니다. (Token Mismatch)" });
    }

    const key = normalize(data.key);
    const pin = String(data.pin || "");
    const machineId = data.machineId;
    const version = data.version || "1.0.0";
    const userInfo = data.userInfo || {};
    
    // 1. 라이선스 키 찾기
    const values = ws.getDataRange().getValues();
    let rowIndex = -1;
    let rowData = null;
    
    for (let i = 1; i < values.length; i++) {
      if (normalize(values[i][COL.KEY - 1]) === key) {
        rowIndex = i + 1;
        rowData = values[i];
        break;
      }
    }
    
    if (!rowData) return response({ valid: false, message: "존재하지 않는 라이선스 키입니다." });
    
    // 2. PIN 번호 검증
    const savedPin = String(rowData[COL.PIN - 1] || "");
    if (savedPin !== pin) return response({ valid: false, message: "PIN 번호가 일치하지 않습니다." });
    
    // 3. 상태 확인
    const status = rowData[COL.STATUS - 1] || "";
    if (status !== "ACTIVE" && status !== "PENDING") {
      return response({ valid: false, message: "라이선스 상태가 유효하지 않습니다: " + status });
    }
    
    const savedMachineId = String(rowData[COL.MACHINE_ID - 1] || "").trim();
    const savedContact = normalize(rowData[COL.COMPANY] === userInfo.contact ? rowData[COL.COMPANY] : (rowData[3] || "")); // 시트 구조에 따라 조정 필요
    // 참고: rowData 인덱스는 0부터 시작하므로 COL - 1
    const actualSavedContact = normalize(rowData[3]); // 연락처가 4번째 열(Index 3)에 있다고 가정
    const providedContact = normalize(userInfo.contact || "");
    const currentTimestamp = new Date();
    
    // ---------------------------------------------------------
    // 4. 지능형 자가 마이그레이션 로직 (v4.6 최적화)
    // ---------------------------------------------------------
    
    // [Case A] 서버에 머신 ID가 없는 경우 (최초 등록)
    if (!savedMachineId) {
      updateLicenseRow(ws, rowIndex, machineId, userInfo, version, "ACTIVE", true, rowData);
      logInstallation(logWs, machineId, "ACTIVATED", "Initial Activation", userInfo, version);
      return response({ valid: true, message: "인증 성공 (신규 기기 등록 완료)" });
    }
    
    // [Case B] 머신 ID가 일치하는 경우 (정상 접속)
    if (savedMachineId === machineId) {
      ws.getRange(rowIndex, COL.CHECKIN).setValue(currentTimestamp);
      ws.getRange(rowIndex, COL.VERSION).setValue(version);
      return response({ valid: true, message: "인증 성공" });
    }
    
    // [Case C] 머신 ID 불일치 (기기 교체 상황)
    // 보안 강화: 등록된 연락처 정보와 일치 여부 확인
    if (providedContact === actualSavedContact && providedContact !== "") {
      
      const lastResetDate = rowData[COL.RESET - 1] ? new Date(rowData[COL.RESET - 1]) : null;
      const diffDays = lastResetDate ? (currentTimestamp - lastResetDate) / (1000 * 60 * 60 * 24) : 999;
      
      // 쿨다운 적용 (7일) - RESET 칸 기준
      if (diffDays >= 7 || !lastResetDate) {
        updateLicenseRow(ws, rowIndex, machineId, userInfo, version, "ACTIVE", true, rowData);
        logInstallation(logWs, machineId, "UPGRADED", "Hardware Upgrade (Auto-Update)", userInfo, version);
        return response({ valid: true, message: "하드웨어 변경이 감지되어 기기 정보를 갱신했습니다." });
      } else {
        const waitDays = Math.ceil(7 - diffDays);
        return response({ valid: false, message: "최근에 기기를 변경하셨습니다. " + waitDays + "일 후에 다시 시도하거나 관리자에게 문의하세요." });
      }
    }
    
    // [Case D] 정보 불일치
    return response({ valid: false, message: "다른 기기에서 사용 중이거나, 등록된 연락처 정보와 일치하지 않습니다." });
    
  } catch (e) {
    return response({ valid: false, message: "서버 오류: " + e.toString() });
  }
}

function updateLicenseRow(ws, rowIndex, machineId, userInfo, version, status, isReset, rowData) {
  const currentTimestamp = new Date();
  ws.getRange(rowIndex, COL.MACHINE_ID).setValue(machineId);
  ws.getRange(rowIndex, COL.STATUS).setValue(status);
  ws.getRange(rowIndex, COL.CHECKIN).setValue(currentTimestamp);
  ws.getRange(rowIndex, COL.VERSION).setValue(version);
  
  // 기존 데이터가 비어있을 때만 클라이언트 정보로 업데이트
  const savedName = String(rowData ? rowData[COL.USER_INFO - 1] : "").trim();
  const savedCompany = String(rowData ? rowData[COL.COMPANY - 1] : "").trim();
  
  if (!savedName) {
    ws.getRange(rowIndex, COL.USER_INFO).setValue((userInfo.name || "") + " (" + (userInfo.position || "직책없음") + ")");
  }
  if (!savedCompany && userInfo.company) {
    ws.getRange(rowIndex, COL.COMPANY).setValue(userInfo.company);
  }
  
  // 실제 기기 변경(Reset)이 일어날 때만 RESET 칸 업데이트
  if (isReset) {
    ws.getRange(rowIndex, COL.RESET).setValue(currentTimestamp);
  }
}

function response(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function normalize(s) {
  return String(s || "").trim().toLowerCase().replace(/[\s-]/g, "");
}

function logInstallation(logWs, machineId, action, result, userInfo, version) {
  if (!logWs) return;
  const timestamp = new Date();
  logWs.appendRow([
    timestamp, 
    userInfo.company || "", 
    userInfo.name || "", 
    userInfo.contact || "", 
    machineId, 
    action, 
    result, 
    "", 
    version
  ]);
}
