/**
 * [LicenseGuard Server v6.8 - Perfect Consolidation for EzImpo]
 * - 설치 로그(InstallLogs) 열 밀림 완벽 수정 및 10컬럼 구조 정합성 유지
 * - [신규] 인스톨/로그인(STARTUP) 시 Licenses 시트의 해당 기기 최신 접속시간(I열) 및 버젼(L열) 자동 동기화 기능 통합
 * - 제품정보(Products) 시트에서 최신 버전을 실시간으로 동적으로 조회하도록 개선 (getLatestVersionFromSheet)
 * - 기기 변경 시 연락처 검증 및 7일 재등록 제한(COOLDOWN) 안내 UX 강화
 * - 대표님 시트 컬럼 순서(A~R) 100% 반영 및 보안 서명 규격(HMAC-SHA256) 호환성 유지
 * - ※ 본 구글 시트는 오직 EzImpo 전용 시트이며, EzPrintWork 시트와 완전히 분리되어 작동합니다.
 */

var _SECRET = "ez_impo_sec_2026_gold"; 
var COOLDOWN_DAYS = 7;

function doGet(e) { return doPost(e); }

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let data = {};
  try {
    data = tryExtractData(e);
    logToDebugSheet(ss, data, e);
    
    const action = data.action || "verify";
    if (action === "checkStatusAndVersion") return handleStatusAndVersion_V68(ss, data);
    if (action === "request_purchase") return handlePurchaseRequest_V68(ss, data);
    if (action === "log_startup") return handleLogStartup_V68(ss, data);
    if (action === "sms_inbound") return handleSmsInbound_V68(ss, data);
    
    return handleVerify_V68(ss, data); 
  } catch (err) {
    return response({ result: "error", msg: "서버 오류: " + err.toString() });
  }
}

// 요청 데이터 파싱
function tryExtractData(e) {
  let d = { _rawContents: "" };
  if (e && e.postData && e.postData.contents) {
    d._rawContents = e.postData.contents;
    try { 
      const parsed = JSON.parse(e.postData.contents); 
      for (let key in parsed) { d[key] = parsed[key]; }
    } catch (err) { d._parseError = err.toString(); }
  }
  if (e && e.parameter) { 
    for (let k in e.parameter) { if (!d[k]) d[k] = e.parameter[k]; } 
  }
  return d;
}

// 1. 설치 및 스타트업 로그 기록 + 라이선스 실시간 정보 동기화 통합
function handleLogStartup_V68(ss, data) {
  const logWs = ss.getSheetByName("InstallLogs") || ss.insertSheet("InstallLogs");
  const nowStr = Utilities.formatDate(new Date(), "GMT+9", "yyyy-MM-dd HH:mm:ss");
  
  const company = data.company || "-";
  const name = data.userName || data.name || "-";
  const contact = formatContact(data.contact);
  const mid = String(data.machineId || "").trim();
  const act = data.status || "STARTUP";
  const ip = data.ip || "-";
  const ver = data.version || data.progVer || "-";
  const pname = data.progName || "EzImpo"; // EzImpo 시트이므로 기본값 고정

  // 시트 컬럼 순서: 1.날짜, 2.업체명, 3.사용자명, 4.연락처, 5.기기ID, 6.동작, 7.결과, 8.IP, 9.버전, 10.제품명
  const rowData = [
    nowStr,
    q(company),
    q(name),
    q(contact),
    q(mid),
    act,
    "Success",
    ip,
    q(ver),
    pname
  ];
  
  logWs.appendRow(rowData);
  
  // [강력 추천 기능] 스타트업 시 해당 사용자의 Licenses 시트 최신 접속일 및 실시간 버전 정보 자동 업그레이드
  const licWs = ss.getSheetByName("Licenses");
  if (licWs && mid !== "") {
    try {
      const values = licWs.getDataRange().getValues();
      const cleanMid = mid.toLowerCase();
      const cleanPname = String(pname).trim().toLowerCase().replace(/\s/g, "");
      
      for (let i = 1; i < values.length; i++) {
        const rowMid = String(values[i][4] || "").trim().toLowerCase(); // E열 (index 4)
        const rowProd = String(values[i][10] || "").trim().toLowerCase().replace(/\s/g, ""); // K열 (index 10)
        
        if (rowMid === cleanMid && (rowProd === cleanPname || !rowProd || cleanPname === "")) {
          licWs.getRange(i + 1, 9).setValue(nowStr); // I열 (Last Check-in)
          if (ver && ver !== "-") {
            licWs.getRange(i + 1, 12).setValue(q(ver)); // L열 (Version)
          }
          break;
        }
      }
    } catch (e) {
      // 라이선스 동기화 오류 시 조용히 넘김 (스타트업 로그 로깅 자체를 방해하지 않음)
    }
  }
  
  return response({ result: "success", status: "active" });
}

// Products 시트에서 제품 최신 버전 동적 조회
function getLatestVersionFromSheet(ss, progName) {
  try {
    const ws = ss.getSheetByName("Products");
    if (!ws) return "3.7.2";
    const data = ws.getDataRange().getValues();
    const searchName = String(progName || "EzImpo").toLowerCase();
    for (let i = 1; i < data.length; i++) {
      const nameInSheet = String(data[i][1]).toLowerCase();
      if (nameInSheet.indexOf(searchName) !== -1) return String(data[i][2]);
    }
  } catch (e) {}
  return "3.7.2";
}

// 2. 버전 및 서버 구동 여부 확인
function handleStatusAndVersion_V68(ss, data) {
  const latest = getLatestVersionFromSheet(ss, data.progName || "EzImpo");
  return response({ 
    result: "success", licenseValid: true, status: "active",
    latestVersion: latest, 
    downloadUrl: "https://naver.me/Fm3SGglJ", 
    notice: "v6.8 서버 정상 가동 중 (최신: " + latest + ")" 
  });
}

// 3. 라이선스 검증 및 쿨다운 기기 초기화
function handleVerify_V68(ss, data) {
  const ws = ss.getSheetByName("Licenses");
  if (!ws) return response({ result: "error", msg: "Licenses 시트 없음" });
  
  const key = normalize(data.key);
  const pin = String(data.pin || "");
  const mid = String(data.machineId || "").trim();
  const appVersion = data.version || data.progVer || data.v || "Unknown";
  const pname = data.progName || "EzImpo";
  
  let u = { 
    company: data.userInfo?.company || data.company || "", 
    name: data.userInfo?.name || data.userName || data.name || "", 
    contact: data.userInfo?.contact || data.contact || "" 
  };
  
  const values = ws.getDataRange().getValues();
  let r = -1; let rd = null;
  for (let i = 1; i < values.length; i++) {
    if (normalize(values[i][0]) === key) { r = i + 1; rd = values[i]; break; }
  }
  
  if (!rd) return response({ result: "error", code: "KEY_NOT_FOUND", msg: "미등록 키" });
  if (String(rd[1]) !== pin && String(rd[1]) !== "") return response({ result: "error", code: "PIN_MISMATCH" });

  const savedMid = String(rd[4] || "").trim();         // E열 (5번열)
  const lastResetDate = rd[9] ? new Date(rd[9]) : null;  // J열 (10번열)
  const providedContact = normalize(u.contact);
  const savedContact = normalize(rd[15]);              // P열 (16번열)

  // 1) 동일 기기에서 재인증 또는 최초 인증 등록
  if (!savedMid || savedMid === mid) {
    updateLicenseRow_V68(ws, r, mid, u, appVersion, "ACTIVE", pname, !savedMid, pin);
    return response({ 
      result: "success", valid: true, status: "active",
      productName: pname || rd[10] || "EzImpo",
      expiryDate: rd[5] || "평생소장",
      data: { key: key, company: u.company || rd[2], version: appVersion }
    });
  }

  // 2) 기기 변경 검증 (연락처 번호 일치 시 7일 쿨타임 제한 후 승인)
  if (providedContact !== "" && providedContact === savedContact) {
    const diffDays = lastResetDate ? (new Date() - lastResetDate) / (1000 * 60 * 60 * 24) : 999;
    if (diffDays >= COOLDOWN_DAYS) {
      updateLicenseRow_V68(ws, r, mid, u, appVersion, "ACTIVE", pname, true, pin);
      return response({ 
        result: "success", valid: true, status: "active", 
        msg: "기기 정보 갱신 완료",
        productName: pname || rd[10] || "EzImpo",
        expiryDate: rd[5] || "평생소장",
        data: { key: key, company: u.company || rd[2], version: appVersion }
      });
    } else {
      const waitDays = Math.ceil(COOLDOWN_DAYS - diffDays);
      return response({ 
        result: "error", 
        code: "DEVICE_MISMATCH", 
        msg: "다른 기기에서 사용 중입니다. (기기 재등록은 7일마다 가능하며, " + waitDays + "일 남았습니다.)" 
      });
    }
  }
  return response({ result: "error", code: "DEVICE_MISMATCH", msg: "다른 기기에서 사용 중입니다." });
}

// 라이선스 테이블의 각 컬럼 업데이트 (A~R열 엄격 매칭)
function updateLicenseRow_V68(ws, r, mid, u, ver, st, pname, isReset, pin) {
  const nowStr = Utilities.formatDate(new Date(), "GMT+9", "yyyy-MM-dd HH:mm:ss");
  const currentPin = ws.getRange(r, 2).getValue();
  
  if (String(currentPin) === "" && pin) ws.getRange(r, 2).setValue(q(pin)); // B열 (PIN)
  if (u.company) ws.getRange(r, 3).setValue(q(u.company));                 // C열 (CompanyName)
  if (u.name) ws.getRange(r, 4).setValue(q(u.name));                       // D열 (UserName)
  ws.getRange(r, 5).setValue(q(mid));                                     // E열 (MachineID)
  ws.getRange(r, 7).setValue(st);                                         // G열 (Status)
  ws.getRange(r, 9).setValue(nowStr);                                     // I열 (Last Check-in)
  if (isReset) ws.getRange(r, 10).setValue(nowStr);                        // J열 (Last Reset)
  ws.getRange(r, 11).setValue(pname || "EzImpo");                          // K열 (ProductName)
  if (ver && ver !== "Unknown") ws.getRange(r, 12).setValue(q(ver));        // L열 (Version)
  if (u.contact) ws.getRange(r, 16).setValue(q(formatContact(u.contact))); // P열 (Contact Info)
}

// 4. 구매 요청 처리 (관리자 앱 스키마와 100% 정합)
function handlePurchaseRequest_V68(ss, data) {
  let ws = ss.getSheetByName("Order");
  if (!ws) {
    ws = ss.insertSheet("Order");
    ws.appendRow(['날짜', '업체명', '입금자', '연락처', '기기ID', '버전', '상태', 'ID', '제품명']);
  }
  const now = new Date();
  const nowStr = Utilities.formatDate(now, "GMT+9", "yyyy-MM-dd HH:mm:ss");
  
  let u = data.userInfo || { company: data.company, depositor: data.depositor || data.userName || "User", contact: data.contact };
  const userName = String(u.depositor || u.name || "User").trim();
  const machineId = String(data.machineId || "-").trim();
  
  const datePart = Utilities.formatDate(now, "GMT+9", "yyyyMMddHHmm");
  const reqId = "req-" + datePart + "-" + userName + "-" + machineId;
  const pname = data.progName || "EzImpo"; 

  // 관리자 앱(Order 시트)의 9개 컬럼 스키마
  const row = [
    nowStr,                     // 1. 날짜 (createdAt)
    q(u.company || "-"),        // 2. 업체명 (companyName)
    q(userName),                // 3. 입금자 (name)
    q(formatContact(u.contact)),// 4. 연락처 (contact)
    q(machineId),               // 5. 기기ID (machineId)
    q(data.version || "-"),     // 6. 버전 (version)
    "PENDING",                  // 7. 상태 (status) -> PENDING 상태여야 신규 알림에 뜸
    q(reqId),                   // 8. ID (id)
    q(pname)                    // 9. 제품명 (productName)
  ];
  
  ws.appendRow(row);
  
  // (하위 호환성용 구버전 구매요청 시트 유지)
  const oldWs = ss.getSheetByName("PurchaseRequests");
  if (oldWs) {
    oldWs.appendRow([nowStr, q(u.company), q(userName), q(u.contact), pname, q(data.version), q(machineId), "WAITING"]);
  }

  return response({ result: "success", msg: "요청이 정상적으로 접수되었습니다." });
}

// [공통 도우미 함수]
function signPayload(payload) { 
  try { 
    const data = JSON.stringify(payload); 
    const signature = Utilities.computeHmacSignature(Utilities.MacAlgorithm.HMAC_SHA_256, data, _SECRET); 
    return signature.map(e => ("0" + (e & 0xFF).toString(16)).slice(-2)).join(""); 
  } catch (e) { return "hash_error"; } 
}

function response(o) { 
  const finalOutput = { ...o, payload: o, hash: signPayload(o) }; 
  return ContentService.createTextOutput(JSON.stringify(finalOutput)).setMimeType(ContentService.MimeType.JSON); 
}

// 휴대폰/연락처 번호 자동 포맷팅
function formatContact(s) {
  if (!s) return "";
  var d = String(s).replace(/\D/g, "");
  if (d.length === 11) return d.replace(/(\d{3})(\d{4})(\d{4})/, "$1-$2-$3");
  if (d.length === 10) {
    if (d.startsWith("02")) return d.replace(/(\d{2})(\d{4})(\d{4})/, "$1-$2-$3");
    return d.replace(/(\d{3})(\d{3})(\d{4})/, "$1-$2-$3");
  }
  return s;
}

// 스프레드시트의 0 누락 방지 접두사(') 추가
function q(v) { 
  if (!v) return ""; 
  var s = String(v).trim(); 
  return (s.indexOf("'") === 0) ? s : "'" + s; 
}

// 검색 및 비교 최적화를 위한 텍스트 정규화
function normalize(s) { 
  if (!s) return ""; 
  return String(s).trim().toLowerCase().replace(/[^a-z0-9가-힣]/g, ""); 
}

// 디버그 보조 로그
function logToDebugSheet(ss, data, e) {
  try {
    let ws = ss.getSheetByName("DebugLogs");
    if (!ws) {
      ws = ss.insertSheet("DebugLogs");
      ws.appendRow(["Timestamp", "Action", "MachineId", "Status", "RawData"]);
    }
    const status = data._parseError ? "PARSE_ERROR" : "OK";
    ws.appendRow([Utilities.formatDate(new Date(), "GMT+9", "yyyy-MM-dd HH:mm:ss"), data.action || "verify", data.machineId || "-", status, data._rawContents || JSON.stringify(data)]);
  } catch (err) {}
}

// [NEW] 스마트폰 앱으로부터 수신한 문자(INBOUND)를 스프레드시트에 저장하는 웹훅 게이트웨이
function handleSmsInbound_V68(ss, data) {
  let ws = ss.getSheetByName("SmsLogs");
  if (!ws) {
    ws = ss.insertSheet("SmsLogs");
    ws.appendRow(['날짜', '연락처', '대화내용', '구분', '상태', '라이선스ID']);
  }
  
  const nowStr = Utilities.formatDate(new Date(), "GMT+9", "yyyy-MM-dd HH:mm:ss");
  const rawContact = String(data.from || data.contact || "").trim();
  const cleanContact = rawContact.replace(/\D/g, "");
  const content = String(data.content || data.text || "").trim();
  
  if (cleanContact === "" || content === "") {
    return response({ result: "error", msg: "필수 파라미터 누락 (from 또는 content)" });
  }
  
  // Licenses 시트에서 이 연락처를 소유한 사용자의 licenseId를 찾아 매핑
  let licenseId = "";
  const licWs = ss.getSheetByName("Licenses");
  if (licWs) {
    try {
      const values = licWs.getDataRange().getValues();
      const targetContact = cleanContact;
      for (let i = 1; i < values.length; i++) {
        const rowContact = String(values[i][15] || "").replace(/\D/g, ""); // P열 (index 15)
        const rowLicenseId = String(values[i][16] || "").trim();           // Q열 (index 16)
        
        if (rowContact === targetContact && rowContact !== "") {
          licenseId = rowLicenseId || String(values[i][0]).trim(); // Q열이 비어있으면 A열(License Key)을 대체 ID로 사용
          break;
        }
      }
    } catch (e) {
      // 라이선스 매핑 실패 시 공백 처리
    }
  }
  
  // SmsLogs 시트에 기록: 날짜, 연락처, 대화내용, 구분, 상태, 라이선스ID
  const rowData = [
    nowStr,
    q(cleanContact),
    content,
    "INBOUND",
    "SUCCESS",
    licenseId
  ];
  
  ws.appendRow(rowData);
  return response({ result: "success", msg: "SMS가 시트에 정상 기록되었습니다.", licenseId: licenseId });
}
